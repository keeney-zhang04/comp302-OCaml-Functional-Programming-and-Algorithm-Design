let binomial n k = 
  let rec factorial a = 
    if a > 0 then a * factorial(a-1) 
    else 1 
  in
  factorial n / ((factorial k) * (factorial (n-k)))

let rec lucas_helper params acc1 acc2=
  if params = 0 then acc1
  else if params = 1 then acc2
  else lucas_helper (params-1) acc2 (acc2+acc1)
