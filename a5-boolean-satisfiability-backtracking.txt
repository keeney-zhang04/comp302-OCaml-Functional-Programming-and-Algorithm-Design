let rec backtrack vars assignment formula =
  match vars with
  | [] -> if eval assignment formula then assignment else raise Unsatisfiable_formula
  | v :: vs ->
      (* Try assigning true to v *)
      (try backtrack vs (Variable_map.add v true assignment) formula with Unsatisfiable_formula ->
        (* If assigning true fails, try false *)
         backtrack vs (Variable_map.add v false assignment) formula)
      
let xy_truth_asgn : truth_assignment = Variable_map.empty

let rec backtrack_cps vars assignment formula return fail =
  match vars with
  | [] ->
      if eval assignment formula then
        return assignment
      else
        fail ()
  | v :: vs ->
      backtrack_cps vs (Variable_map.add v true assignment) formula return (fun () ->
          backtrack_cps vs (Variable_map.add v false assignment) formula return fail)

let rec find_sat_assignment_cps (formula : formula)
    (return : truth_assignment -> 'r) (fail : unit -> 'r) : 'r =
  let vars = collect_variables formula in
  backtrack_cps vars Variable_map.empty formula return fail
